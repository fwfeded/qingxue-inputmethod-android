cmake_minimum_required(VERSION 3.22)
project(qingstudy_text LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_MTMD OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(CMAKE_SHARED_LINKER_FLAGS "${CMAKE_SHARED_LINKER_FLAGS} -Wl,-z,max-page-size=16384")
set(LLAMA_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../../.tools/text-models/llama.cpp-b11750" CACHE PATH "已校验的 llama.cpp 源码")
add_subdirectory(${LLAMA_SOURCE} llama)
add_library(qingstudy_text SHARED text.cpp)
target_include_directories(qingstudy_text PRIVATE ${LLAMA_SOURCE}/vendor)
target_link_libraries(qingstudy_text PRIVATE llama)
